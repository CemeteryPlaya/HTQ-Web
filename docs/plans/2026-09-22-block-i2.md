# Блок I.2 — хвосты блока I и поддомены: план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** закрыть хвосты блока I и перевести компании на короткие поддомены,
после чего блоки I + I.2 можно выкатить одним окном.

**Architecture:** компания получает короткий псевдоним (`Company.subdomain`),
middleware резолвит метку хоста в слаг, фронт приземляет вход с голого домена
на экран выбора компании. Параллельно закрываются находки финального ревью
блока I: сужение на одной ручке, привязка именных ролей к компании, удаление
экрана на снятой эвристике, остатки `hr_level`, слепые пятна сторожей.

**Tech Stack:** Django 5.2.7 (hand-rolled `htqweb.http.api_view`, не DRF),
Postgres со схемой на компанию, React + Vite + vitest, pytest-django.

**Spec:** [docs/plans/2026-09-22-block-i2-spec.md](2026-09-22-block-i2-spec.md)

```python
    def clean(self):
        """Метка адреса уникальна по ВСЕМ хостам, а не только по своей колонке.

        Один хост обязан резолвиться в одну компанию, поэтому псевдоним не
        может совпасть со слагом другой компании, а слаг — с чужим
        псевдонимом. Своя же пара (slug == subdomain) — один и тот же хост,
        это допустимо.
        """
        super().clean()
        errors = {}
        if self.subdomain:
            if self.subdomain in RESERVED_SUBDOMAINS:
                errors["subdomain"] = (
                    f"{self.subdomain!r} — зарезервированная метка: "
                    f"{', '.join(sorted(RESERVED_SUBDOMAINS))}."
                )
            elif (Company.objects.filter(slug=self.subdomain)
                  .exclude(pk=self.pk).exists()):
                errors["subdomain"] = (
                    f"{self.subdomain!r} — слаг другой компании; один хост не "
                    "может вести в две компании."
                )
        if self.slug and (Company.objects.filter(subdomain=self.slug)
                          .exclude(pk=self.pk).exists()):
            errors["slug"] = (
                f"{self.slug!r} — псевдоним другой компании; один хост не "
                "может вести в две компании."
            )
        if errors:
            raise ValidationError(errors)
```

``company_create --subdomain`` или правка компании.

from django.db import migrations

SUBDOMAINS = {
    "hi-tech-qazaqstan": "htq",
    "hi-tech-systems": "hts",
    "kazakhstan-engineering-group": "keg",
    "hi-tech-group": "group",
}


def set_subdomains(apps, schema_editor):
    Company = apps.get_model("companies", "Company")
    for slug, label in SUBDOMAINS.items():
        Company.objects.filter(slug=slug, subdomain__isnull=True).update(subdomain=label)


def unset_subdomains(apps, schema_editor):
    Company = apps.get_model("companies", "Company")
    for slug, label in SUBDOMAINS.items():
        Company.objects.filter(slug=slug, subdomain=label).update(subdomain=None)


class Migration(migrations.Migration):
    dependencies = [("companies", "0005_company_subdomain")]
    operations = [migrations.RunPython(set_subdomains, unset_subdomains)]

    """Адрес компании для ссылок, уходящих наружу: https://<метка>.<корень>.

    Корень берётся из ``PUBLIC_BASE_URL`` (https://htq.group → htq.group).
    Нужен там, где ссылка ведёт на страницу, читающую таблицы КОМПАНИИ:

  company.subdomain || company.slug;

/** Перейти в другую компанию, сохранив текущий путь, query-параметры и hash. */
export const switchCompany = (company: { slug: string; subdomain?: string | null }): void => {
  const { host, pathname, search, hash, protocol } = window.location;
  const tail = parentDomain(host).replace(/^\./, '');
  const port = host.includes(':') ? `:${host.split(':')[1]}` : '';
  window.location.assign(
    `${protocol}//${hostLabelOf(company)}.${tail}${port}${pathname}${search}${hash}`,
  );
};

import re

from django.db import migrations

CUSTOM_CODE = re.compile(r"^hr-custom-(?P<slug>.+)-(?P<position_id>\d+)$")


def fill(apps, schema_editor):
    Role = apps.get_model("access", "Role")
    for role in Role.objects.filter(code__startswith="hr-custom-",
                                    company_slug__isnull=True):
        match = CUSTOM_CODE.match(role.code)
        if match:

            role.save(update_fields=["company_slug"])


def unfill(apps, schema_editor):

```python
_CLASS_RE = re.compile(r"^class\s+(\w+)\s*\(\s*(?:[\w.]+\s*,\s*)*(?:ApiView|CompaniesView|AccessView)\b")
_METHOD_RE = re.compile(r"^(\s+)(?:async\s+)?def\s+(get|post|put|patch|delete)\s*\(")


def _iter_undecorated_api_methods(text: str):
    """Методы-ручки в классах-вьюхах, над которыми нет ни одного декоратора.

    Сторож полноты читает вызовы ``api_view(``; метод CBV без декоратора он
    не видел вовсе — то есть ручка без гейта была для него невидимой
    (блок I.2, R7).
    """
    lines = text.splitlines()
    current_class = None
    for idx, line in enumerate(lines):
        klass = _CLASS_RE.match(line)
        if klass:
            current_class = klass.group(1)
            continue
        if line and not line[0].isspace():
            current_class = None
        method = _METHOD_RE.match(line)
        if method and current_class:
            previous = lines[idx - 1].strip() if idx else ""
            if not previous.startswith("@"):
                yield idx + 1, f"{current_class}.{method.group(2)}"
```
